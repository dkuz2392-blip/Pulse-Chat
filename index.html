async function initAuth() {
try {
if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
await signInWithCustomToken(auth, __initial_auth_token);
} else {
await signInAnonymously(auth);

}

// FIRESTORE REALTIME SYNC
function subscribeToPublicMessages() {
if (!authenticatedUser) return;
if (unsubscribeMessages) unsubscribeMessages();
const messagesCollection = collection(db, 'artifacts', appId, 'public', 'data', 'messages');
unsubscribeMessages = onSnapshot(messagesCollection, (snapshot) => {
snapshot.docChanges().forEach((change) => {
if (change.type === "added") {
const data = change.doc.data();
if (data && data.message && data.chatCode) {
handleCloudIncomingMessage(data);
}
}
});
}, (error) => {
console.warn("Firestore snapshot listener alert:", error.message);
});
}
function handleCloudIncomingMessage(data) {
if (!window.currentUser) return;
if (data.senderCode === window.currentUser.code) return; // Skip own messages
let chat = window.chats.find(c => c.code === data.chatCode || (c.code && c.code.toLowerCase() === data.chatCode.toLowerCase()));
if (!chat && data.receiverCode === window.currentUser.code) {
chat = {
id: 'friend_' + Date.now(),
name: data.senderName || 'Chat (' + data.senderCode + ')',
type: 'friend',
avatar: (data.senderName || 'U').charAt(0).toUpperCase(),
code: data.senderCode,
status: '● Live Realtime',
messages: []
};
window.chats.push(chat);
}
if (chat) {
if (!chat.messages) chat.messages = [];
if (!chat.messages.some(m => m.id === data.message.id)) {
chat.messages.push(data.message);
window.saveChats();
window.renderChats();
if (window.activeChatId === chat.id) {
window.renderMessages();
} else {
window.showToast(window.t('newMsgFrom') + (data.senderName || 'partner'));
}
}
}
}

receiverCode: receiverCode,
senderCode: window.currentUser.code,
senderName: window.currentUser.name,

});
} catch (err) {

}
};

window.sendMessage = function() {
const input = document.getElementById('messageInput');
const text = input.value.trim();
if (!text && !window.selectedAttachment) return;
const newMessage = {
id: Date.now(),
sender: window.currentUser.name,
text: text,
file: window.selectedAttachment,
time: getFormattedTime()
};
let chat = window.chats.find(c => c.id === window.activeChatId);
if (!chat) {
chat = window.chats[0];
window.activeChatId = chat.id;
}
if (!chat.messages) chat.messages = [];
chat.messages.push(newMessage);
window.saveChats();
window.renderMessages();
window.renderChats();
// Send Realtime Cloud Message
if (chat.type === 'friend' && chat.code && chat.code !== '#saved_notes') {
window.sendCloudMessage(chat.code, chat.code, newMessage);
}
if (chat.type === 'bot') {
setTimeout(() => {
chat.messages.push({
id: Date.now() + 1,
sender: chat.name,
text: `${window.t('botResponsePrefix')} ("${text || 'file'}"). ${window.t('botResponseSuffix')}`,
time: getFormattedTime()
});
window.saveChats();
if (window.activeChatId === chat.id) window.renderMessages();
window.renderChats();
}, 400);
}
input.value = '';
window.clearAttachment();
};
window.renderMessages = function() {
const container = document.getElementById('messagesContainer');
if (!container) return;
container.innerHTML = '';
const chat = window.chats.find(c => c.id === window.activeChatId);
if (!chat || !chat.messages || chat.messages.length === 0) {
container.innerHTML = `<div class="flex flex-col items-center justify-center h-full text-slate-500 space-y-2"> <i class="fa-regular fa-comments text-4xl opacity-20"></i> <p class="text-xs text-center">${window.t('noMessages')}</p> </div>`;
return;
}
chat.messages.forEach(msg => {
const isMe = msg.sender === window.currentUser?.name;
const wrapper = document.createElement('div');
wrapper.className = `flex flex-col ${isMe ? 'items-end' : 'items-start'} space-y-1`;
let contentHtml = '';
if (msg.text) {
contentHtml += `<p class="leading-relaxed whitespace-pre-wrap break-words">${escapeHtml(msg.text)}</p>`;
}
if (msg.file) {
if (msg.file.type && msg.file.type.startsWith('image/')) {
contentHtml += `<img src="${msg.file.data}" class="max-w-xs rounded-lg mt-1 border border-slate-700">`;
} else {
contentHtml += `<a href="${msg.file.data}" download="${msg.file.name}" class="flex items-center space-x-2 mt-1 p-2 bg-slate-800 rounded-lg text-cyan-400 text-xs"> <i class="fa-solid fa-file-arrow-down"></i> <span class="underline truncate">${msg.file.name}</span> </a>`;
}
}
wrapper.innerHTML = `<div class="max-w-[85%] md:max-w-[70%] px-3.5 py-2 rounded-2xl text-xs md:text-sm ${isMe ? 'bg-cyan-600 text-white rounded-br-none shadow-md' : 'bg-slate-800 text-slate-100 rounded-bl-none border border-slate-700/50'}"> ${contentHtml} <div class="text-[9px] ${isMe ? 'text-cyan-200' : 'text-slate-400'} text-right mt-1"> ${msg.time} </div> </div>`;
container.appendChild(wrapper);
});
container.scrollTop = container.scrollHeight;
};
